from __future__ import annotations
import os
from datetime import datetime
from typing import Optional
import psycopg2
from psycopg2.extras import RealDictCursor
from pydantic import BaseModel, Field


class Tools:
    class Valves(BaseModel):
        DATABASE_URL: str = Field(
            default=os.getenv(
                "SCHEDULED_TASKS_DATABASE_URL",
                "postgresql://postgres:zFns3MZAQNnZ2UVj3q41j1kJn0ORdfJ9qNjHU7skR7ev50Ugi9y7aGOsrFlBbQPs@5.78.120.77:5434/soren_openwebui",
            ),
            description="Cadena de conexión a Postgres para scheduled_tasks.",
        )

    def __init__(self):
        self.valves = self.Valves()

    def _ensure_table(self, cur) -> None:
        cur.execute(
            """
            CREATE TABLE IF NOT EXISTS scheduled_tasks (
                id SERIAL PRIMARY KEY,
                prompt TEXT NOT NULL,
                run_at TIMESTAMPTZ NOT NULL,
                recurrence TEXT,
                recurrence_interval_hours INTEGER,
                recurrence_end TIMESTAMPTZ,
                notify BOOLEAN NOT NULL DEFAULT FALSE,
                status TEXT NOT NULL DEFAULT 'pending',
                created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
                executed_at TIMESTAMPTZ,
                last_error TEXT,
                last_response_preview TEXT
            );
            """
        )
        cur.execute(
            """
            DO $$
            BEGIN
                IF NOT EXISTS (
                    SELECT 1 FROM pg_constraint WHERE conname = 'scheduled_tasks_status_check'
                ) THEN
                    ALTER TABLE scheduled_tasks
                    ADD CONSTRAINT scheduled_tasks_status_check
                    CHECK (status IN ('pending','running','done','failed','completed'));
                END IF;
            END
            $$;
            """
        )
        cur.execute(
            """CREATE INDEX IF NOT EXISTS scheduled_tasks_pending_idx ON scheduled_tasks (status, run_at);"""
        )
        cur.execute(
            """ALTER TABLE scheduled_tasks ADD COLUMN IF NOT EXISTS recurrence TEXT;"""
        )
        cur.execute(
            """ALTER TABLE scheduled_tasks ADD COLUMN IF NOT EXISTS recurrence_interval_hours INTEGER;"""
        )
        cur.execute(
            """ALTER TABLE scheduled_tasks ADD COLUMN IF NOT EXISTS recurrence_end TIMESTAMPTZ;"""
        )

    def _serialize(self, val):
        if isinstance(val, datetime):
            return val.isoformat()
        return val

    async def schedule_task(
        self,
        prompt: str,
        run_at: str,
        notify: bool = False,
        recurrence: Optional[str] = None,
        recurrence_interval_hours: Optional[int] = None,
        recurrence_end: Optional[str] = None,
    ) -> dict:
        """
        Crea una tarea programada para Soren.

        :param prompt: Instrucción completa que se ejecutará en el futuro.
        :param run_at: Fecha/hora ISO8601 (UTC), ej. 2025-12-05T12:00:00Z.
        :param notify: Si es true, el worker enviará webhook a n8n tras ejecutar.
        :param recurrence: daily | weekly | monthly | None
        :param recurrence_interval_hours: Intervalo en horas (int) para recurrencia.
        :param recurrence_end: ISO8601 UTC o None (recurrencia infinita).
        """
        try:
            datetime.fromisoformat(run_at.replace("Z", "+00:00"))
        except ValueError as exc:
            return {"error": f"run_at inválido: {exc}"}

        if recurrence and recurrence not in {"daily", "weekly", "monthly"}:
            return {"error": "recurrence debe ser daily, weekly o monthly"}

        if recurrence_interval_hours is not None:
            try:
                interval = int(recurrence_interval_hours)
            except (TypeError, ValueError) as exc:
                return {"error": f"recurrence_interval_hours inválido: {exc}"}
            if interval <= 0:
                return {"error": "recurrence_interval_hours debe ser mayor que 0"}
            recurrence_interval_hours = interval

        rec_end_dt: Optional[datetime] = None
        if recurrence_end:
            try:
                rec_end_dt = datetime.fromisoformat(
                    recurrence_end.replace("Z", "+00:00")
                )
            except ValueError as exc:
                return {"error": f"recurrence_end inválido: {exc}"}

        conn = psycopg2.connect(self.valves.DATABASE_URL)
        conn.autocommit = True
        with conn, conn.cursor(cursor_factory=RealDictCursor) as cur:
            self._ensure_table(cur)
            cur.execute(
                """
                INSERT INTO scheduled_tasks (prompt, run_at, notify, status, recurrence, recurrence_interval_hours, recurrence_end)
                VALUES (%s, %s, %s, 'pending', %s, %s, %s)
                RETURNING *;
                """,
                (prompt, run_at, notify, recurrence, recurrence_interval_hours, rec_end_dt),
            )
            row = cur.fetchone()
        conn.close()

        return {"inserted": {k: self._serialize(v) for k, v in row.items()}}

    async def edit_task(
        self,
        task_id: int,
        prompt: Optional[str] = None,
        run_at: Optional[str] = None,
        notify: Optional[bool] = None,
        recurrence: Optional[str] = None,
        recurrence_interval_hours: Optional[int] = None,
        recurrence_end: Optional[str] = None,
    ) -> dict:
        """
        Edita una tarea programada existente.

        :param task_id: ID de la tarea a editar.
        :param prompt: Nueva instrucción (opcional).
        :param run_at: Nueva fecha/hora ISO8601 UTC (opcional).
        :param notify: Nuevo valor de notify (opcional).
        :param recurrence: daily | weekly | monthly | None (opcional).
        :param recurrence_interval_hours: Intervalo en horas (opcional).
        :param recurrence_end: ISO8601 UTC o None (opcional).
        """
        updates = []
        params = []

        if prompt is not None:
            updates.append("prompt = %s")
            params.append(prompt)

        if run_at is not None:
            try:
                datetime.fromisoformat(run_at.replace("Z", "+00:00"))
            except ValueError as exc:
                return {"error": f"run_at inválido: {exc}"}
            updates.append("run_at = %s")
            params.append(run_at)

        if notify is not None:
            updates.append("notify = %s")
            params.append(notify)

        if recurrence is not None:
            if recurrence not in {"daily", "weekly", "monthly", ""}:
                return {"error": "recurrence debe ser daily, weekly, monthly o vacío"}
            updates.append("recurrence = %s")
            params.append(recurrence if recurrence else None)

        if recurrence_interval_hours is not None:
            try:
                interval = int(recurrence_interval_hours)
            except (TypeError, ValueError) as exc:
                return {"error": f"recurrence_interval_hours inválido: {exc}"}
            if interval <= 0:
                return {"error": "recurrence_interval_hours debe ser mayor que 0"}
            updates.append("recurrence_interval_hours = %s")
            params.append(interval)

        if recurrence_end is not None:
            if recurrence_end == "":
                updates.append("recurrence_end = %s")
                params.append(None)
            else:
                try:
                    rec_end_dt = datetime.fromisoformat(
                        recurrence_end.replace("Z", "+00:00")
                    )
                    updates.append("recurrence_end = %s")
                    params.append(rec_end_dt)
                except ValueError as exc:
                    return {"error": f"recurrence_end inválido: {exc}"}

        if not updates:
            return {"error": "No se proporcionaron campos para actualizar"}

        params.append(task_id)

        conn = psycopg2.connect(self.valves.DATABASE_URL)
        conn.autocommit = True
        with conn, conn.cursor(cursor_factory=RealDictCursor) as cur:
            cur.execute(
                f"""
                UPDATE scheduled_tasks
                SET {", ".join(updates)}
                WHERE id = %s AND status = 'pending'
                RETURNING *;
                """,
                params,
            )
            row = cur.fetchone()
        conn.close()

        if not row:
            return {"error": f"Tarea {task_id} no encontrada o ya no está pendiente"}

        return {"updated": {k: self._serialize(v) for k, v in row.items()}}

    async def delete_task(self, task_id: int) -> dict:
        """
        Elimina una tarea programada.

        :param task_id: ID de la tarea a eliminar.
        """
        conn = psycopg2.connect(self.valves.DATABASE_URL)
        conn.autocommit = True
        with conn, conn.cursor(cursor_factory=RealDictCursor) as cur:
            cur.execute(
                """
                DELETE FROM scheduled_tasks
                WHERE id = %s
                RETURNING id, prompt, run_at, status;
                """,
                (task_id,),
            )
            row = cur.fetchone()
        conn.close()

        if not row:
            return {"error": f"Tarea {task_id} no encontrada"}

        return {"deleted": {k: self._serialize(v) for k, v in row.items()}}

    async def list_pending_tasks(self, limit: int = 20) -> dict:
        """
        Lista las tareas programadas pendientes.

        :param limit: Máximo de tareas a devolver (default 20).
        """
        conn = psycopg2.connect(self.valves.DATABASE_URL)
        conn.autocommit = True
        with conn, conn.cursor(cursor_factory=RealDictCursor) as cur:
            self._ensure_table(cur)
            cur.execute(
                """
                SELECT id, prompt, run_at, notify, recurrence, recurrence_interval_hours, recurrence_end, status, created_at
                FROM scheduled_tasks
                WHERE status = 'pending'
                ORDER BY run_at ASC
                LIMIT %s;
                """,
                (limit,),
            )
            rows = cur.fetchall()
        conn.close()

        tasks = [{k: self._serialize(v) for k, v in row.items()} for row in rows]
        return {"pending_tasks": tasks, "count": len(tasks)}
